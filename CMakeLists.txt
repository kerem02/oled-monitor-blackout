cmake_minimum_required(VERSION 3.20)
project(OLEDBlackout VERSION 2.1.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Portable releases must not require a VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(oled_core src/core.cpp)
target_include_directories(oled_core PUBLIC src)
function(oled_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endfunction()
oled_warnings(oled_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(oled_core_tests tests/core_tests.cpp)
  target_link_libraries(oled_core_tests PRIVATE oled_core)
  oled_warnings(oled_core_tests)
  add_test(NAME core COMMAND oled_core_tests)
endif()
if(WIN32)
  enable_language(RC)
  add_executable(OLEDBlackout WIN32 src/main.cpp src/app.cpp src/settings_ui.cpp src/instance.cpp src/platform.cpp resources/app.rc)
  target_include_directories(OLEDBlackout PRIVATE resources)
  target_compile_definitions(OLEDBlackout PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(OLEDBlackout PRIVATE oled_core user32 gdi32 shell32 advapi32 ole32 comctl32 wtsapi32 uuid)
  oled_warnings(OLEDBlackout)
  if(MSVC)
    target_link_options(OLEDBlackout PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
  elseif(MINGW)
    target_compile_options(OLEDBlackout PRIVATE -ffunction-sections -fdata-sections)
    target_compile_options(oled_core PRIVATE -ffunction-sections -fdata-sections)
    target_link_options(OLEDBlackout PRIVATE -Wl,--gc-sections -municode -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
    if(BUILD_TESTING)
      target_link_options(oled_core_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
  endif()
  if(BUILD_TESTING)
    add_executable(oled_platform_tests tests/platform_tests.cpp src/platform.cpp src/instance.cpp)
    target_compile_definitions(oled_platform_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(oled_platform_tests PRIVATE oled_core user32 gdi32 shell32 advapi32 ole32 uuid)
    oled_warnings(oled_platform_tests)
    if(MINGW)
      target_link_options(oled_platform_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME windows_platform COMMAND oled_platform_tests)
  endif()
  install(TARGETS OLEDBlackout RUNTIME DESTINATION .)
  install(FILES README.md LICENSE CHANGELOG.md DESTINATION .)
endif()
